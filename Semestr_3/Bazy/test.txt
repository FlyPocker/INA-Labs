# 2.3 nie uzywajac join wyswietl tytuly filmow i ich dlugosci dla filmow o 4 najmniejszych roznych dlugosciach w calej bazie, wyniki posortuj malejaco po dlugosci
SELECT title, length 
FROM film 
WHERE length IN (
    SELECT length 
    FROM (
        SELECT DISTINCT length 
        FROM film 
        ORDER BY length ASC 
        LIMIT 4
    ) AS subquery
)
ORDER BY length DESC;

# 3.4 wypisz imiona i nazwiska aktorów, którzy nie wystapili w zadnym filmie, ktorego tytul zaczyna sie na litere C
SELECT first_name, last_name 
FROM actor 
WHERE actor_id NOT IN (
    SELECT fa.actor_id 
    FROM film_actor fa 
    JOIN film f ON fa.film_id = f.film_id 
    WHERE f.title LIKE 'C%'
);

# 3.5 wypisz bez powtorzen tytuly filmow z kategorii Action i ich opisy, uwzglednij tylko te filmy, ktore wypozyczone byly pomiedzy 20 a 30 czerwca 2005 oraz w ktorych opisie nie wystepuje slowo Drama
SELECT DISTINCT f.title, f.description 
FROM film f 
JOIN film_category fc ON f.film_id = fc.film_id 
JOIN category c ON fc.category_id = c.category_id 
JOIN inventory i ON f.film_id = i.film_id 
JOIN rental r ON i.inventory_id = r.inventory_id 
WHERE c.name = 'Action' 
  AND r.rental_date BETWEEN '2005-06-20' AND '2005-06-30' 
  AND f.description NOT LIKE '%Drama%';

# 3.6 wypisz tytuły filmów z kategorii Action, których data wypożyczenia (z dokładnością do dnia) jest inna niż data dokonania płatności
SELECT DISTINCT f.title 
FROM film f 
JOIN film_category fc ON f.film_id = fc.film_id 
JOIN category c ON fc.category_id = c.category_id 
JOIN inventory i ON f.film_id = i.film_id 
JOIN rental r ON i.inventory_id = r.inventory_id 
JOIN payment p ON r.rental_id = p.rental_id 
WHERE c.name = 'Action' 
  AND DATE(r.rental_date) != DATE(p.payment_date);

# 4.7 policz ilu klientów wypozyczylo wiecej filmow niz klient o emailu MARY.SMITH@sakilacustomer.org
SELECT COUNT(*) AS customers_count 
FROM (
    SELECT customer_id 
    FROM rental 
    GROUP BY customer_id 
    HAVING COUNT(rental_id) > (
        SELECT COUNT(r2.rental_id) 
        FROM rental r2 
        JOIN customer c2 ON r2.customer_id = c2.customer_id 
        WHERE c2.email = 'MARY.SMITH@sakilacustomer.org'
    )
) AS sub;

# 4.8 wypisz bez powtorzen imiona i nazwiska klientow, ktorych srednia oplata za wypozyczony film jest nizsza niz srednia oplata dokonana przez wszystkich 30 lipca 2005
SELECT c.first_name, c.last_name 
FROM customer c 
JOIN payment p ON c.customer_id = p.customer_id 
GROUP BY c.customer_id, c.first_name, c.last_name 
HAVING AVG(p.amount) < (
    SELECT AVG(amount) 
    FROM payment 
    WHERE DATE(payment_date) = '2005-07-30'
);

# 4.9 Wypisz wszystkie pary różnych identyfikatorów aktorów, którzy wystąpili razem w więcej niż dwóch filmach. Każda para powinna występować co najwyżej raz. Jeśli występuje para (X, Y), to nie wypisuj pary (Y, X).
SELECT fa1.actor_id AS actor_1, fa2.actor_id AS actor_2 
FROM film_actor fa1 
JOIN film_actor fa2 ON fa1.film_id = fa2.film_id AND fa1.actor_id < fa2.actor_id 
GROUP BY fa1.actor_id, fa2.actor_id 
HAVING COUNT(fa1.film_id) > 2;

# 4.10 wypisz imiona i nazwiska różnych aktorów, którzy wystąpili razem w więcej niż 2 filmach oraz liczbę filmów, w których zagrali razem. Każda para powinna występować co najwyżej raz.
SELECT a1.first_name AS a1_first_name, a1.last_name AS a1_last_name, 
       a2.first_name AS a2_first_name, a2.last_name AS a2_last_name, 
       COUNT(fa1.film_id) AS movies_together 
FROM film_actor fa1 
JOIN film_actor fa2 ON fa1.film_id = fa2.film_id AND fa1.actor_id < fa2.actor_id 
JOIN actor a1 ON fa1.actor_id = a1.actor_id 
JOIN actor a2 ON fa2.actor_id = a2.actor_id 
GROUP BY fa1.actor_id, a1.first_name, a1.last_name, fa2.actor_id, a2.first_name, a2.last_name 
HAVING COUNT(fa1.film_id) > 2;

# 4.11 wypisz nazwiska aktorów, którzy zagrali w większej liczbie horrorów niż filmów akcji
SELECT a.last_name 
FROM actor a 
LEFT JOIN film_actor fa ON a.actor_id = fa.actor_id 
LEFT JOIN film_category fc_h ON fa.film_id = fc_h.film_id 
    AND fc_h.category_id = (SELECT category_id FROM category WHERE name = 'Horror') 
LEFT JOIN film_category fc_a ON fa.film_id = fc_a.film_id 
    AND fc_a.category_id = (SELECT category_id FROM category WHERE name = 'Action') 
GROUP BY a.actor_id, a.last_name 
HAVING COUNT(fc_h.film_id) > COUNT(fc_a.film_id);
